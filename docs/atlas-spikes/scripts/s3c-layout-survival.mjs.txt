import { createRequire } from "node:module";
import path from "node:path";
const require = createRequire(path.resolve("noop.js"));
const { chromium } = require("playwright-core");
const BASE = "http://localhost:3101";
const browser = await chromium.launch({ executablePath: process.env.CHROME ?? "/opt/pw-browsers/chromium-1194/chrome-linux/chrome", args: ["--no-sandbox", "--disable-background-networking", "--no-first-run"] });
const page = await (await browser.newContext()).newPage();
await page.goto(`${BASE}/atlas?lat=45.07&lon=7.69`);
await page.waitForFunction(() => window.__atlas);
const snap = () => page.evaluate(() => ({ search: location.search.replace(/&delay=\d+/, ""), shell: document.querySelector("#shell")?.dataset.mount ?? null, n: document.querySelector("#shell")?.dataset.n ?? null, content: document.querySelector("#content")?.textContent ?? null, atlasError: !!document.querySelector("#atlas-error"), rootError: !!document.querySelector("#error-boundary") }));
const out = [];
await page.evaluate(() => window.__atlas.bump());
out.push(["start", await snap()]);
await page.evaluate(() => window.__atlas.replace("/atlas?lat=41.1&lon=7.69&delay=400"));
await page.waitForFunction(() => document.querySelector("#content")?.textContent?.includes("41.1"));
out.push(["after a good replace", await snap()]);
await page.evaluate(() => window.__atlas.replace("/atlas?lat=41.2&lon=7.69&delay=300&fail=1"));
await page.waitForFunction(() => document.querySelector("#atlas-error") || document.querySelector("#error-boundary"), null, { timeout: 8000 }).catch(() => out.push(["timeout waiting for any error boundary", null]));
await page.waitForTimeout(600);
out.push(["after a throwing replace", await snap()]);
await page.evaluate(() => window.__atlas.replace("/atlas?lat=41.3&lon=7.69&delay=200"));
await page.waitForFunction(() => document.querySelector("#content")?.textContent?.includes("41.3"), null, { timeout: 6000 }).catch(() => {});
await page.waitForTimeout(500);
out.push(["recovery by the next replace (6 s allowed)", await snap()]);
await page.reload();
await page.waitForTimeout(800);
out.push(["after a full reload of the same URL", await snap()]);
console.log(JSON.stringify(out, null, 1));
await browser.close();
