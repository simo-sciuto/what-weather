// S7 (and the software side of S2): Mapbox GL JS camera semantics, events, limits and queryRenderedFeatures,
// with an inline blank style and inline GeoJSON. No Mapbox service is contacted and no token is used.
// Run: node run.mjs <path to a scratch install holding mapbox-gl and playwright-core>
import { createRequire } from "node:module";
import path from "node:path";

const root = path.resolve(process.argv[2] ?? ".");
const require = createRequire(path.join(root, "noop.js"));
const { chromium } = require("playwright-core");
const gl = path.join(root, "node_modules/mapbox-gl/dist");
const CHROME = process.env.CHROME ?? "/opt/pw-browsers/chromium-1194/chrome-linux/chrome";

const browser = await chromium.launch({
  executablePath: CHROME,
  args: ["--no-sandbox", "--use-angle=swiftshader", "--enable-unsafe-swiftshader", "--ignore-gpu-blocklist", "--enable-webgl"],
});
const page = await browser.newPage({ viewport: { width: 800, height: 600 } });
const errors = [];
page.on("pageerror", (e) => errors.push(String(e).slice(0, 200)));
// a real origin is needed for the web worker blob (about:blank gives an opaque origin and the GeoJSON worker never answers)
await page.route("https://spike.local/**", (route) => route.fulfill({ contentType: "text/html", body: `<!doctype html><html><body style="margin:0"><div id="map" style="width:800px;height:600px"></div></body></html>` }));
await page.goto("https://spike.local/");
await page.addStyleTag({ path: path.join(gl, "mapbox-gl.css") });
await page.addScriptTag({ path: path.join(gl, "mapbox-gl.js") });

const out = { mapboxGl: await page.evaluate(() => mapboxgl.version), chromium: browser.version() };

// ---- build the map with an inline style and test data
out.setup = await page.evaluate(async () => {
  const line = { type: "Feature", properties: { kind: "river", name: "Riv" }, geometry: { type: "LineString", coordinates: [[7.0, 45.0], [8.4, 45.14]] } };
  const lake = { type: "Feature", properties: { kind: "lake", name: "Lago" }, geometry: { type: "Polygon", coordinates: [[[7.6, 45.0], [7.8, 45.0], [7.8, 45.1], [7.6, 45.1], [7.6, 45.0]]] } };
  const peak = { type: "Feature", properties: { kind: "peak", name: "Cima", ele: 4634 }, geometry: { type: "Point", coordinates: [7.9, 45.12] } };
  const style = {
    version: 8,
    sources: { d: { type: "geojson", data: { type: "FeatureCollection", features: [line, lake, peak] } } },
    layers: [
      { id: "bg", type: "background", paint: { "background-color": "#345" } },
      { id: "river", type: "line", source: "d", filter: ["==", ["get", "kind"], "river"], paint: { "line-color": "#9cf", "line-width": 2 } },
      { id: "river-hidden", type: "line", source: "d", filter: ["==", ["get", "kind"], "river"], layout: { visibility: "none" }, paint: { "line-color": "#f00", "line-width": 2 } },
      { id: "river-zero", type: "line", source: "d", filter: ["==", ["get", "kind"], "river"], paint: { "line-color": "#f00", "line-opacity": 0, "line-width": 14 } },
      { id: "lake", type: "fill", source: "d", filter: ["==", ["get", "kind"], "lake"], paint: { "fill-color": "#6cf" } },
      { id: "peak", type: "circle", source: "d", filter: ["==", ["get", "kind"], "peak"], paint: { "circle-radius": 5, "circle-color": "#fff" } },
      { id: "late", type: "circle", source: "d", filter: ["==", ["get", "kind"], "peak"], minzoom: 11, paint: { "circle-radius": 3, "circle-color": "#f0f" } },
    ],
  };
  const t0 = performance.now();
  const map = new mapboxgl.Map({ container: "map", style, center: [7.687, 45.07], zoom: 8, accessToken: "pk.unused-no-service-is-contacted", testMode: false, attributionControl: false, trackResize: false });
  window.map = map;
  const loaded = new Promise((res) => map.once("load", () => res(performance.now() - t0)));
  const failed = new Promise((res) => map.once("error", (e) => res("error: " + String(e.error && e.error.message).slice(0, 120))));
  const r = await Promise.race([loaded, failed, new Promise((res) => setTimeout(() => res("timeout"), 15000))]);
  window.log = [];
  return { loadMs: typeof r === "number" ? Math.round(r) : r, webgl: !!document.querySelector("canvas") };
});

if (!out.setup.webgl || typeof out.setup.loadMs !== "number") {
  out.blocked = "map did not load in this environment";
  console.log(JSON.stringify({ out, errors }, null, 2));
  await browser.close();
  process.exit(0);
}

const px = (fn, ...a) => page.evaluate(fn, ...a);
const settle = (ms) => page.waitForTimeout(ms);

// ---- A. getCenter and padding
out.A_center_padding = await px(() => {
  const m = window.map, r = {};
  m.jumpTo({ center: [7.687, 45.07], zoom: 8, padding: { top: 0, bottom: 0, left: 0, right: 0 } });
  const c0 = m.getCenter(); r.noPadding = { getCenter: [c0.lng, c0.lat], unprojectCanvasCentre: Object.values(m.unproject([400, 300]).toArray()) };
  m.jumpTo({ padding: { top: 0, bottom: 200, left: 0, right: 0 } });
  const c1 = m.getCenter();
  r.bottom200 = {
    getCenter: [c1.lng, c1.lat],
    unprojectCanvasCentre: m.unproject([400, 300]).toArray(),
    unprojectPaddedCentre: m.unproject([400, 200]).toArray(),
    padding: m.getPadding(),
  };
  r.getCenterEqualsPaddedCentre = Math.abs(c1.lat - m.unproject([400, 200]).lat) < 1e-9;
  r.getCenterEqualsCanvasCentre = Math.abs(c1.lat - m.unproject([400, 300]).lat) < 1e-9;
  m.jumpTo({ padding: { top: 0, bottom: 0, left: 0, right: 0 } });
  return r;
});

// ---- helper: log events with timestamps
await px(() => {
  const m = window.map;
  window.log = [];
  window.t0 = performance.now();
  ["movestart", "move", "moveend", "dragstart", "dragend", "zoomstart", "zoomend", "pitchstart", "pitchend", "rotatestart", "rotateend", "idle"].forEach((n) =>
    m.on(n, (e) => window.log.push({ t: Math.round(performance.now() - window.t0), n, user: !!(e && e.originalEvent), type: e && e.originalEvent ? e.originalEvent.type : null, moving: m.isMoving() })),
  );
});
const summary = async () => px(() => {
  const l = window.log, by = {};
  l.forEach((e) => { by[e.n] = (by[e.n] || 0) + 1; });
  const first = (n) => l.find((e) => e.n === n), last = (n) => l.slice().reverse().find((e) => e.n === n);
  return { counts: by, movestart: first("movestart"), moveend: last("moveend"), idle: last("idle"), firstMove: first("move") };
});

// ---- B2. user drag: slow stop (no inertia) and fast flick (inertia)
async function drag(steps, stepMs, endPause) {
  await px(() => { window.log = []; window.t0 = performance.now(); });
  await page.mouse.move(400, 300);
  await page.mouse.down();
  for (let i = 1; i <= steps; i++) { await page.mouse.move(400 + i * (80 / steps) * 1.0, 300 + i * (20 / steps)); if (stepMs) await settle(stepMs); }
  if (endPause) await settle(endPause);
  const tUp = await px(() => Math.round(performance.now() - window.t0));
  await page.mouse.up();
  await settle(2500);
  const s = await summary();
  s.mouseUpAt = tUp;
  s.moveendAfterMouseUpMs = s.moveend ? s.moveend.t - tUp : null;
  return s;
}
out.B2_slow_drag_stop = await drag(8, 40, 400);
await px(() => window.map.jumpTo({ center: [7.687, 45.07], zoom: 8 })); await settle(200);
out.B2_fast_flick = await drag(6, 4, 0);
await px(() => window.map.jumpTo({ center: [7.687, 45.07], zoom: 8 })); await settle(200);

// ---- E. queryRenderedFeatures (software side of S2): tolerance, hidden and zero-opacity layers, minzoom, pitch
await px(() => window.map.jumpTo({ center: [7.7, 45.07], zoom: 9, pitch: 0, bearing: 0 }));
await settle(500);
out.E_qrf = await px(() => {
  const m = window.map, r = {};
  const names = (fs) => Array.from(new Set(fs.map((f) => f.layer.id))).sort();
  // find the pixel of a point on the river
  const onLine = m.project([7.7, 45.0 + (0.7 / 1.4) * 0.14 + 0.0]);
  // the river runs from (7.0,45.0) to (8.4,45.14); at lon 7.7 lat = 45.07
  const p = m.project([7.7, 45.07]);
  r.projectedRiverPixel = [Math.round(p.x), Math.round(p.y)];
  const q = (dx, dy, opts) => names(m.queryRenderedFeatures([p.x + dx, p.y + dy], opts));
  r.pointOnLine = q(0, 0);
  r.offsets = {};
  [2, 4, 6, 8, 10, 16].forEach((d) => { r.offsets["+" + d + "px_y"] = q(0, d); });
  r.layersOnlyRiver = q(0, 0, { layers: ["river"] });
  r.hiddenLayerIncluded = q(0, 0, { layers: ["river-hidden"] });
  r.zeroOpacityIncluded = q(0, 0, { layers: ["river-zero"] });
  r.zeroOpacityWide7px = q(0, 7, { layers: ["river-zero"] });
  r.zeroOpacityWide16px = q(0, 16, { layers: ["river-zero"] });
  const box = (d) => names(m.queryRenderedFeatures([[p.x - d, p.y - d], [p.x + d, p.y + d]], { layers: ["river"] }));
  r.box12 = box(12);
  const lakeP = m.project([7.7, 45.05]); r.lakePoint = names(m.queryRenderedFeatures([lakeP.x, lakeP.y]));
  const peakP = m.project([7.9, 45.12]);
  r.peakAtZoom9 = names(m.queryRenderedFeatures([peakP.x, peakP.y]));
  r.peakFeatureProps = (m.queryRenderedFeatures([peakP.x, peakP.y], { layers: ["peak"] })[0] || {}).properties || null;
  r.peakTolerance = [3, 6, 9, 14].map((d) => [d, names(m.queryRenderedFeatures([peakP.x + d, peakP.y], { layers: ["peak"] })).length]);
  return r;
});

// ---- F. cost of re-creating the map (what the backdrop does at every place change)
out.F_recreate = await px(async () => {
  const times = [];
  for (let i = 0; i < 3; i++) {
    const style = { version: 8, sources: {}, layers: [{ id: "bg", type: "background", paint: { "background-color": "#123" } }] };
    const t0 = performance.now();
    const holder = document.createElement("div"); holder.style.cssText = "width:400px;height:300px;position:absolute;left:-9999px"; document.body.appendChild(holder);
    const m2 = new mapboxgl.Map({ container: holder, style, center: [0, 0], zoom: 2, accessToken: "pk.unused", attributionControl: false });
    await new Promise((res) => m2.once("load", res));
    times.push(Math.round(performance.now() - t0));
    m2.remove(); holder.remove();
  }
  return { constructToLoadMs: times };
});
